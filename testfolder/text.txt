til funksjonen er lengde, stilpoeng og kpunkt.

"""
antar lengde er float eller int. antar kpunkt er float eller int. antar elementene
"""
def beregnPoeng(lengde, stilpoeng, kpunkt):
    
    #Lengdepoeng
    lengde_over_kpunkt = lengde - kpunkt
    lengdepoeng = lengde_over_kpunkt * 1.8 #positiv hvis over kpunkt og negativ hv
    if lengdepoeng >= 0:
        lengdepoeng += 60 #Legger til 60 poeng hvis kpunkt er nådd
        
    #Stilpoeng
    stilpoeng.sort()
    stilpoeng.pop(-1)
    stilpoeng.pop(0) #Fjerner laveste og høyest element. kunne også ha brukt del s
        verdien
    stilpoeng_sum = sum(stilpoeng)
    
    return lengdepoeng + stilpoeng_sum

Innen kryptografi er cæsarchiffer, også kjent som Caesars chiffer, skiftchiffer, Cæsars
kode og cæsarskift, en av de enkleste og mest kjente krypteringsteknikker. Det er en
type substitusjonschiffer der hver bokstav i klarteksten erstattes med en annen bokstav et gitt
antall steg lenger ut i alfabetet. Med et skift på tre steg erstattes D med A, E med B, F med C og
så videre. Metoden har navn etter Julius Cæsar som benyttet denne koden til å kommunisere
med sine generaler.
 
Lag funksjonen krypter(tekst,skift) som krypterer en tekst med cæsarchiffer. Inputvariabelen
skift forteller hvor mange steg en bokstav forflytter seg i alfabetet.
 
Vår funksjon er noe forenklet og skal kun ta hensyn til bokstaver i det norske alfabetet.
Funksjonen skal returnere den krypterte teksten med store bokstaver, og alle mellomrom skal
være fjernet.
 
Eksempelvis vil krypter("Kryptering er gøy",1) returnere den krypterte
teksten LSZQUFSJOHFSHÅZ

"""
I første avsnitt av oppgaven sies det at positiv skift er til venstre (i alfabetet
    sies det at positiv skift er til høyre. Jeg har valgt at positiv skift er til 
"""
def krypter(tekst, skift): #Antar skift alltid er int og at tekst alltid er str
    alfabet = "ABCDEFGHIJKLMNOPQRSTUVWXYZÆØÅ"
    
    tekst = tekst.replace(" ", "").upper() #fjerner mellomrom og gjør om til store
    nytekst = "" #kryptert tekst
    
    for char in tekst:
        charindex = int(alfabet.find(char))
        nytekst += alfabet[charindex - skift] #dersom skiften er forbi a (index 0)
            det bli hentet fra slutten av alfabetet.
    
    return nytekst
